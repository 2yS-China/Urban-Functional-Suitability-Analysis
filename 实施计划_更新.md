    # 4. Function REQUIRES Rule

### 2.3.4 推理引擎实现

**创建 `backend/kg/reasoning_engine.py`**：

```python
from typing import List, Dict, Optional
from .knowledge_graph_builder import SpatialKnowledgeGraph, RelationType

class ReasoningChain:
    """推理链数据结构"""
    def __init__(self):
        self.steps: List[Dict] = []
    
    def add_step(self, source, relation, target, evidence):
        """添加推理步骤"""
        self.steps.append({
            'source': source,
            'relation': relation,
            'target': target,
            'evidence': evidence
        })
    
    def to_natural_language(self) -> str:
        """转为自然语言描述"""
        explanation = []
        for step in self.steps:
            rel_desc = self._relation_to_chinese(step['relation'])
            explanation.append(
                f"{step['source'].properties.get('name', step['source'].id)} "
                f"{rel_desc} "
                f"{step['target'].properties.get('name', step['target'].id)}"
            )
            if step['evidence']:
                explanation.append(f"（{step['evidence']}）")
        return "，".join(explanation)
    
    def _relation_to_chinese(self, rel_type):
        mapping = {
            RelationType.NEAR: "靠近",
            RelationType.SERVES_BY: "由...服务",
            RelationType.CONNECTS_TO: "连接到",
            RelationType.REQUIRES: "要求",
            RelationType.SATISFIES: "满足"
        }
        return mapping.get(rel_type, str(rel_type))


class ReasoningEngine:
    """知识推理引擎"""
    
    def __init__(self, kg: SpatialKnowledgeGraph):
        self.kg = kg
    
    def explain_suitability(self, unit_id: str, function: str, 
                           max_depth: int = 3) -> ReasoningChain:
        """
        解释适宜性：为什么单元X适合功能Y
        
        推理路径示例：
        单元X -> NEAR -> 地铁站M -> SERVES_BY -> L2号线 -> CONNECTS_TO -> CBD
              -> SATISFIES -> 居住规则(800m内有地铁)
        """
        chain = ReasoningChain()
        
        # 1. 找到单元到功能的适宜性关系
        suitability_rel = self._find_relation(
            unit_id, f"function_{function}", RelationType.SUITABLE_FOR
        )
        if not suitability_rel:
            return chain
        
        # 2. 找到功能的关键要求
        function_node = self.kg.get_node(f"function_{function}")
        requirements = self.kg.get_outgoing_relations(
            function_node.id, RelationType.REQUIRES
        )
        
        # 3. 对每个要求进行推理
        for req_rel in requirements:
            rule_id = req_rel.target
            rule_node = self.kg.get_node(rule_id)
            
            # 3.1 查找满足该规则的证据链
            evidence_chain = self._find_evidence_chain(
                unit_id, rule_node, max_depth
            )
            
            if evidence_chain:
                for step in evidence_chain:
                    chain.add_step(**step)
        
        return chain
    
    def _find_evidence_chain(self, start_node_id: str, 
                            target_rule: Node, 
                            max_depth: int) -> List[Dict]:
        """
        广度优先搜索，找到从单元到规则满足的证据链
        
        示例路径：
        unit_123 --NEAR(500m)--> poi_地铁站_001 --SERVES_BY--> metro_line_L2 
                 --CONNECTS_TO--> area_CBD
        """
        from collections import deque
        
        queue = deque([(start_node_id, [])])
        visited = {start_node_id}
        
        while queue and len(queue) < 1000:  # 防止无限循环
            current_id, path = queue.popleft()
            
            if len(path) >= max_depth:
                continue
            
            # 获取当前节点的所有出边
            relations = self.kg.get_outgoing_relations(current_id)
            
            for rel in relations:
                target_node = self.kg.get_node(rel.target)
                
                if rel.target not in visited:
                    visited.add(rel.target)
                    new_path = path + [{
                        'source': self.kg.get_node(current_id),
                        'relation': rel.relation_type,
                        'target': target_node,
                        'evidence': self._extract_evidence(rel)
                    }]
                    
                    # 检查是否达到目标规则
                    if self._matches_rule(target_node, target_rule):
                        return new_path
                    
                    queue.append((rel.target, new_path))
        
        return []
    
    def _extract_evidence(self, relation: Relation) -> str:
        """提取关系的证据描述"""
        if relation.relation_type == RelationType.NEAR:
            dist = relation.properties.get('distance', 0)
            return f"距离{dist:.0f}米"
        elif relation.relation_type == RelationType.SERVES_BY:
            return "服务于该线路"
        elif relation.relation_type == RelationType.CONNECTS_TO:
            return "连接到核心区域"
        return ""
    
    def _find_relation(self, source, target, rel_type):
        """查找特定关系"""
        rels = self.kg.get_outgoing_relations(source, rel_type)
        return next((r for r in rels if r.target == target), None)
    
    def _matches_rule(self, node, rule_node):
        """检查节点是否匹配规则"""
        # 实现规则匹配逻辑
        return node.id == rule_node.id or \
               node.properties.get('type') == rule_node.properties.get('required_type')
```

### 2.3.5 集成到自然语言生成器

**修改 `backend/kg/nlg_generator.py`**：

```python
from .reasoning_engine import ReasoningEngine

class NaturalLanguageGenerator:
    def __init__(self, kg: SpatialKnowledgeGraph):
        self.kg = kg
        self.reasoning_engine = ReasoningEngine(kg)  # 新增
    
    def answer_why(self, unit_id: str, function: str) -> str:
        """回答：为什么单元X适合功能Y"""
        
        # 使用推理引擎生成因果链
        reasoning_chain = self.reasoning_engine.explain_suitability(
            unit_id, function, max_depth=4
        )
        
        if not reasoning_chain.steps:
            return self._fallback_explanation(unit_id, function)
        
        # 生成分层解释
        answer = f"单元 {unit_id} 适合{function}功能，具体原因如下：\n\n"
        
        # 第一层：直接证据
        answer += "**区位优势**：\n"
        answer += reasoning_chain.to_natural_language() + "\n\n"
        
        # 第二层：规则满足
        answer += "**规则符合性**：\n"
        satisfied_rules = self._get_satisfied_rules(unit_id, function)
        for rule in satisfied_rules:
            answer += f"- {rule['description']}: {rule['status']}\n"
        
        # 第三层：关联影响
        answer += "\n**关联优势**：\n"
        indirect_benefits = self._get_indirect_benefits(reasoning_chain)
        for benefit in indirect_benefits:
            answer += f"- {benefit}\n"
        
        return answer
    
    def _get_indirect_benefits(self, chain: ReasoningChain) -> List[str]:
        """从推理链提取间接优势"""
        benefits = []
        for step in chain.steps:
            if step['relation'] == RelationType.CONNECTS_TO:
                target_name = step['target'].properties.get('name', '')
                benefits.append(f"通过地铁可快速到达{target_name}")
        return benefits
```
